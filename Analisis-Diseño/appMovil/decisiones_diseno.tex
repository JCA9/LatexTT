\subsubsection{Por que Flutter}

\begin{table}[H]
\centering
\small
\begin{tabularx}{\textwidth}{|>{\raggedright\arraybackslash}p{4.2cm}|X|}
\hline
\textbf{Criterio} & \textbf{Justificación} \\
\hline
Desarrollo multiplataforma & Un código base para Android e iOS reduce tiempo 40-50\%. Expansión a iOS planeada en v2.0 \\
\hline
Hot Reload & Velocidad de desarrollo aumenta significativamente según estudios de Google \cite{flutter_performance} \\
\hline
Material Design nativo & No requiere librerías adicionales, garantiza consistencia visual \\
\hline
Curva de aprendizaje & Equipo con experiencia en Java/Kotlin, transición a Dart fue fluida \\
\hline
Rendimiento & Compilación a código nativo ARM garantiza 60 FPS \\
\hline
Comunidad & 160,000+ paquetes en pub.dev \\
\hline
\end{tabularx}
\caption{Justificación de Flutter}
\end{table}

\textbf{Alternativas descartadas:}
\begin{itemize}
    \item \textbf{React Native:} Menor rendimiento en animaciones complejas
    \item \textbf{Nativo (Kotlin):} Duplicar esfuerzo para iOS
    \item \textbf{Xamarin:} Menor soporte de comunidad
\end{itemize}

\textbf{Razones de selección:}
\begin{enumerate}

    \item Personalización completa de colores, estilos, tooltips
    \item Documentación clara con ejemplos

\subsubsection{Por que SharedPreferences}

\textbf{Ventajas para prototipo:}
\begin{itemize}
    \item Simplicidad de implementación
    \item Suficiencia para datos de sesión (email, token, config)
    \item Velocidad de acceso
    \item Sin overhead de configuración
\end{itemize}

\textbf{Limitaciones reconocidas:}
\begin{itemize}
    \item No escala para datos complejos
    \item Sin queries relacionales
    \item \textbf{Migración planeada:} SQLite en v2.0 para caché local
\end{itemize}
